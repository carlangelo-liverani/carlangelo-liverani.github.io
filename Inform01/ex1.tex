\large
Si calcolino, se esistono, I seguenti limiti:
\begin{enumerate}
\item $\lim\limits_{n\to\infty} n\sin 2\pi n$
\item $\lim\limits_{n\to\infty}\frac {\sin n}{n}$
\item $\lim\limits_{n\to\infty} \frac{n^2}{n-1}-n$
\item $\lim\limits_{n\to\infty} \frac {n^{\frac 13}}{\sqrt{n}}$
\item $\lim\limits_{n\to\infty} \frac 1{2^n}$
\item $\lim\limits_{n\to\infty} 2^{\frac 1n}$
\item $\lim\limits_{n\to\infty} \frac {n^2}{3^n}$
\item $\lim\limits_{n\to\infty} \frac{\sin 2^n}{n}$
\item $\lim\limits_{n\to\infty} \frac
{2^n}{n}\left[\sqrt{4^n+n}-\sqrt{4^n+1}\right]$ 
\item $\lim\limits_{n\to\infty}a_n$ dove $a_n$ \`e la successione:
$a_1=1/2$, $a_2=1/2+1/4$, $a_3=1/2+1/4+1/8$, $a_4=1/2+1/4+1/8+1/16$ e
cos\`\i\ via.
\end{enumerate}
