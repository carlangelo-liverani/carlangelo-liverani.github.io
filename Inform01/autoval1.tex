\centerline{\Large Prima Prova di Autovalutazione}
\vskip1cm
\begin{enumerate}
\item Si trovino tutti i numeri per cui
\[
\sin x\leq \sin 2x.
\]
\item Dimostrare che per ogni $x,y\in\Q$
\[
|x+y|\leq |x|+|y|.
\]

\item Dimostrare che se $p\in\N$ \`e primo, allora $\sqrt p$ \`e irrazionale.

\item Trovare il massimo comune divisore tra 1428 e 935.

\item Trovare tutti i divisori di 391.

\item Scrivere la frazione continua associata a $\sqrt 3$.

\end{enumerate}
