\begin{enumerate}
\item Tra tutti i triangoli iscritti nel cerchio di raggio uno e aventi
per lato un diametro si trovi quello di area massima.
\item Si studi il limite
\[
\lim_{n\to\infty}\int_{\frac 1n}^{\frac 2n}\frac{\cos x}{\sqrt x} dx.
\]
\item Si determini quante soluzioni negative ha l'equazione
\[
x^5-5x-1=0.
\]
\item Si studi la convergenza della serie
\[
\sum_{n=0}^\infty
\frac{2^n}{n^3}\left(\sqrt{4^n+n}-\sqrt{4^n+1}\right).
\]
\item Si dica quale dei seguenti numeri \`e maggiore:
\[
12^{211};\quad 100!;\quad 1000000^9.
\]
\item Trovare per quali valori di $x\in{\mathbb R}$ \`e soddisfatta la disu\-guaglianza
\[
\sin x\leq \sin 2x.
\]
\end{enumerate}
